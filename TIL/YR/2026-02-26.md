# 2. AI eye tracking 기능 구현
- 동공 움직임에 따른 마우스 움직임 구현
```py
import cv2
import mediapipe as mp
import pyautogui
import numpy as np
import time

BaseOptions = mp.tasks.BaseOptions
FaceLandmarker = mp.tasks.vision.FaceLandmarker
FaceLandmarkerOptions = mp.tasks.vision.FaceLandmarkerOptions
VisionRunningMode = mp.tasks.vision.RunningMode

MODEL_PATH = 'face_landmarker.task'

class EyeMouseController:
    def __init__(self):
        options = FaceLandmarkerOptions(
            base_options=BaseOptions(model_asset_path=MODEL_PATH),
            running_mode=VisionRunningMode.VIDEO,
            num_faces=1,
            min_face_detection_confidence=0.5,
            min_tracking_confidence=0.5,
        )
        self.face_landmarker = FaceLandmarker.create_from_options(options)

        # 화면 크기
        self.screen_w, self.screen_h = pyautogui.size()

        # 캘리브레이션 데이터
        self.calibration = None

    def calibrate(self, frames=100):
        """
        5점 캘리브레이션:
        화면 4 코너 + 중앙을 순서대로 보게 함
        """
        calib_points = [
            ((100, 100),                                    "화면 왼쪽 위 가장자리를 보세요"),
            ((self.screen_w-100, 100),                      "화면 오른쪽 위 가장자리를 보세요"),
            ((100, self.screen_h-100),                      "화면 왼쪽 아래 가장자리를 보세요"),
            ((self.screen_w-100, self.screen_h-100),        "화면 오른쪽 아래 가장자리를 보세요"),
            ((self.screen_w//2, self.screen_h//2),          "화면 가운데를 보세요"),
        ]

        calib_data = []

        for point, label in calib_points:
            print(f"{label} (3초간 유지)")

            gaze_samples = []
            for _ in range(frames):
                gaze = self.get_gaze_point()
                if gaze:
                    gaze_samples.append(gaze)

            if gaze_samples:
                avg_gaze = np.mean(gaze_samples, axis=0)
                calib_data.append((avg_gaze, point))  # point는 (x, y) 튜플

        self.calibration = calib_data

    def get_gaze_point(self):
        """
        현재 동공 위치로 시선 좌표 추정
        """
        ret, frame = self.cap.read()
        if not ret:
            return None

        timestamp_ms = int(time.time() * 1000)
        mp_image = mp.Image(image_format=mp.ImageFormat.SRGB, data=cv2.cvtColor(frame, cv2.COLOR_BGR2RGB))
        results = self.face_landmarker.detect_for_video(mp_image, timestamp_ms)

        if not results.face_landmarks:
            return None

        landmarks = results.face_landmarks[0]

        # 왼쪽 눈 동공 (랜드마크 468)
        left_pupil = landmarks[468]
        # 오른쪽 눈 동공 (랜드마크 473)
        right_pupil = landmarks[473]

        # 평균 위치
        pupil_x = (left_pupil.x + right_pupil.x) / 2
        pupil_y = (left_pupil.y + right_pupil.y) / 2

        return (pupil_x, pupil_y)

    def map_to_screen(self, gaze_point):
        """
        동공 좌표 → 화면 좌표 변환
        """
        if not self.calibration:
            screen_x = int(gaze_point[0] * self.screen_w)
            screen_y = int(gaze_point[1] * self.screen_h)
            return screen_x, screen_y

        # TODO: Polynomial Regression 등으로 보간
        screen_x = int(gaze_point[0] * self.screen_w)
        screen_y = int(gaze_point[1] * self.screen_h)
        return screen_x, screen_y

    def detect_blink(self):
        """
        눈 깜빡임 감지 (EAR 알고리즘)
        """
        pass

    def run(self):
        """
        메인 루프
        """
        self.cap = cv2.VideoCapture(0)

        # 캘리브레이션
        print("캘리브레이션 시작...")
        self.calibrate()

        print("시선 마우스 시작!")

        while True:
            gaze = self.get_gaze_point()

            if gaze:
                screen_x, screen_y = self.map_to_screen(gaze)

                # 마우스 부드럽게 이동 (스무딩)
                current_x, current_y = pyautogui.position()
                smooth_x = current_x * 0.7 + screen_x * 0.3
                smooth_y = current_y * 0.7 + screen_y * 0.3

                pyautogui.moveTo(smooth_x, smooth_y)

            if self.detect_blink():
                pyautogui.click()

            # ESC로 종료
            if cv2.waitKey(1) & 0xFF == 27:
                break

        self.cap.release()
        self.face_landmarker.close()

# 실행
controller = EyeMouseController()
controller.run()
```
